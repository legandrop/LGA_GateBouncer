cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(GateBouncerLocalFacts LANGUAGES CXX)
endif()
if(NOT WIN32)
    message(FATAL_ERROR "Local file facts require Windows")
endif()
add_library(gatebouncer_localfacts STATIC LocalFacts.cpp WindowsSignatureBackend.cpp)
target_compile_features(gatebouncer_localfacts PUBLIC cxx_std_17)
target_include_directories(gatebouncer_localfacts PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(gatebouncer_localfacts PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(gatebouncer_localfacts PRIVATE bcrypt)
add_executable(GateBouncerSignatureHelper SignatureHelper.cpp)
target_compile_features(GateBouncerSignatureHelper PRIVATE cxx_std_17)
target_compile_definitions(GateBouncerSignatureHelper PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
if(MSVC)
    set_property(TARGET gatebouncer_localfacts GateBouncerSignatureHelper PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(gatebouncer_localfacts PRIVATE /W4 /permissive-)
    target_compile_options(GateBouncerSignatureHelper PRIVATE /W4 /permissive-)
elseif(MINGW)
    target_compile_options(gatebouncer_localfacts PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(GateBouncerSignatureHelper PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(GateBouncerSignatureHelper PRIVATE -static -municode)
endif()
