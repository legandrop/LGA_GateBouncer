cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(GateBouncerAppIdentity LANGUAGES CXX)
endif()
if(NOT WIN32)
    message(FATAL_ERROR "AppIdentity requires Windows")
endif()
add_library(gatebouncer_appidentity STATIC AppIdentity.cpp WindowsSidFormatter.cpp)
target_compile_features(gatebouncer_appidentity PUBLIC cxx_std_17)
target_include_directories(gatebouncer_appidentity PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(gatebouncer_appidentity PRIVATE advapi32)
if(MSVC)
    target_compile_options(gatebouncer_appidentity PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(gatebouncer_appidentity PRIVATE -Wall -Wextra -Wpedantic)
endif()
