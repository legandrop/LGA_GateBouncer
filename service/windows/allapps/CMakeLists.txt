cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(GateBouncerAllApps LANGUAGES CXX)
endif()
if(NOT TARGET gatebouncer_appidentity)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../src/appidentity" "${CMAKE_CURRENT_BINARY_DIR}/appidentity")
endif()
add_library(gatebouncer_allapps STATIC AllAppsMetadata.cpp AllAppsModel.cpp)
target_compile_features(gatebouncer_allapps PUBLIC cxx_std_17)
target_include_directories(gatebouncer_allapps PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(gatebouncer_allapps PUBLIC gatebouncer_appidentity)
if(MSVC)
    target_compile_options(gatebouncer_allapps PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(gatebouncer_allapps PRIVATE -Wall -Wextra -Wpedantic)
endif()
