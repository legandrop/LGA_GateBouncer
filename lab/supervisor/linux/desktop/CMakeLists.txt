cmake_minimum_required(VERSION 3.20)
project(GateBouncerPrivateDesktop LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Windows MSVC SDK required")
endif()
add_executable(desktop_worker desktop_worker.cpp PrivateDesktopOwner.cpp)
target_compile_features(desktop_worker PRIVATE cxx_std_17)
target_compile_definitions(desktop_worker PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0601)
target_compile_options(desktop_worker PRIVATE /W4 /WX /utf-8)
target_link_libraries(desktop_worker PRIVATE User32 Advapi32 Bcrypt)
