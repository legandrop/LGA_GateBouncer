cmake_minimum_required(VERSION 3.20)
project(GateBouncerP3Guest LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Windows MSVC SDK required")
endif()
add_library(guest_conversion SHARED GuestConversionLauncher.cpp P3Transfer.cpp
    P3OwnedConversionFileSeal.cpp ../files/RetainedFile.cpp)
target_compile_definitions(guest_conversion PRIVATE GB_P3_DLL)
add_executable(conversion_worker conversion_worker.cpp P3Transfer.cpp convert_ndis.cpp)
foreach(target guest_conversion conversion_worker)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0602)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    target_link_libraries(${target} PRIVATE Bcrypt Advapi32)
endforeach()
target_link_libraries(conversion_worker PRIVATE Tdh)
